<!-- FIREBASE -->
<script src="https://www.gstatic.com/firebasejs/10.12.2/firebase-app-compat.js"></script>
<script src="https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore-compat.js"></script>

<script>
const firebaseConfig = {
  apiKey: "PASTE_HERE",
  authDomain: "PASTE_HERE",
  projectId: "PASTE_HERE"
};

firebase.initializeApp(firebaseConfig);
const db = firebase.firestore();

let questions=[
{q:"What is CPT?",o:["Wrong","Critical Pull Time","Wrong"],a:1},
{q:"What is VPM?",o:["Wrong","Mismatch system vs physical","Wrong"],a:1}
];

let i=0,score=0,user="";

function start(){
 user=document.getElementById("loginId").value;
 if(!user){alert("Enter ID");return;}
 document.getElementById("exam").style.display="block";
 next();
}

function next(){
 if(i>=questions.length){finish();return;}
 let q=questions[i];
 document.getElementById("question").innerText=q.q;
 document.getElementById("options").innerHTML=q.o.map((o,idx)=>
 `<div onclick="ans(${idx})">${o}</div>`
 ).join("");
}

function ans(idx){
 if(idx===questions[i].a) score++;
 i++;
 next();
}

function finish(){
 let percent=Math.round((score/questions.length)*100);

 let result={
  loginId:user,
  percent:percent,
  completedAt:new Date().toISOString()
 };

 db.collection("hou2_results").add(result);

 alert("Submitted "+percent+"%");
}

db.collection("hou2_results")
.orderBy("completedAt","desc")
.onSnapshot(snap=>{
 let html="";
 snap.forEach(doc=>{
  let r=doc.data();
  html+=`<div>${r.loginId} - ${r.percent}%</div>`;
 });
 document.getElementById("results").innerHTML=html;
});
</script>
