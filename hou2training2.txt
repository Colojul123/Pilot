        id:q.id,
        cat:q.cat,
        q:q.q,
        a: randomizedOptions.map(x => x.text),
        c: correctIndex
      };
    });
    return randomized;
  }

  function resetQuizState(){
    clearInterval(interval);
    index = 0;
    score = 0;
    answeredCount = 0;
    time = QUESTION_TIME;
    timings = [];
    cheatTriggered = false;
    quizActive = false;
    quizQuestions = [];
  }

  function startQuiz(){
    const user = $("name").value.trim();
    if(!user){
      alert("Please enter login ID or name.");
      $("name").focus();
      return;
    }
    if(questionBank.length < 2){
      alert("Question bank is too small. Add more questions first.");
      return;
    }

    resetQuizState();
    quizQuestions = buildQuizQuestions();
    $("qTotal").textContent = String(quizQuestions.length);
    $("loginCard").classList.add("hidden");
    $("resultCard").classList.add("hidden");
    $("quizCard").classList.remove("hidden");
    quizActive = true;
    loadQuestion();
  }

  function loadQuestion(){
    clearInterval(interval);
    const q = quizQuestions[index];
    if(!q){
      finishQuiz();
      return;
    }

    time = QUESTION_TIME;
    questionStartMs = Date.now();
    $("timer").textContent = String(time);
    $("timerFill").style.width = "100%";
    $("qNumber").textContent = String(index + 1);
    $("qCategory").textContent = q.cat || "General";
    $("answeredCount").textContent = String(answeredCount);
    $("qText").textContent = q.q;

    const labels = ["A", "B", "C", "D"];
    $("options").innerHTML = q.a.map((opt, i) => `
      <button class="option-btn" data-index="${i}">
        <span class="answer-tag">${labels[i] || (i+1) + "."}</span>${escapeHtml(opt)}
      </button>
    `).join("");

    [...$("options").querySelectorAll("button")].forEach(btn => {
      btn.addEventListener("click", () => nextQuestion(Number(btn.dataset.index)));
    });

    interval = setInterval(() => {
      time--;
      $("timer").textContent = String(Math.max(0, time));
      const pct = Math.max(0, (time / QUESTION_TIME) * 100);
      $("timerFill").style.width = pct + "%";
      if(time <= 0){
        nextQuestion(-1);
      }
    }, 1000);
  }

  function nextQuestion(selectedIndex){
    if(!quizActive) return;
    clearInterval(interval);

    const current = quizQuestions[index];
    const spent = Math.min(QUESTION_TIME, Math.max(0, Math.round((Date.now() - questionStartMs) / 1000)));
    timings.push(spent);

    if(selectedIndex === current.c) score++;
    answeredCount++;
    index++;
    if(index >= quizQuestions.length){
      finishQuiz();
      return;
    }
    loadQuestion();
  }

  async function finishQuiz(forcedReason = ""){
    clearInterval(interval);
    quizActive = false;

    $("quizCard").classList.add("hidden");
    $("resultCard").classList.remove("hidden");

    const total = quizQuestions.length || 1;
    const percent = Math.round((score / total) * 100);
    const avgTime = timings.length ? (timings.reduce((a,b)=>a+b,0) / timings.length).toFixed(1) : "0.0";
    const statusMeta = getResultClass(percent);

    $("scoreRing").className = `score-ring ${statusMeta.ring}`;
    $("scoreRing").textContent = percent + "%";
    $("resultBanner").className = `result-banner ${statusMeta.banner}`;
    $("resultBanner").textContent = forcedReason ? "ENDED" : statusMeta.text;
    $("scoreText").textContent = `${score} / ${total} correct`;
    $("resultBreakdown").innerHTML = `
      <div><strong>Name:</strong> ${escapeHtml($("name").value.trim())}</div>
      <div><strong>Department:</strong> ${escapeHtml($("dept").value)}</div>
      <div><strong>Status:</strong> ${forcedReason ? escapeHtml(forcedReason) : statusMeta.text}</div>
      <div><strong>Average time/question:</strong> ${avgTime}s</div>
      <div><strong>Completed:</strong> ${new Date().toLocaleString()}</div>
    `;

    const resultPayload = {
      name: $("name").value.trim(),
      department: $("dept").value,
      score: percent,
      correct: score,
      total,
      avgTime: Number(avgTime),
      status: forcedReason || statusMeta.text,
      time: nowIso()
    };

    await storeResult(resultPayload);
    sendEmailResult(resultPayload);
    await sendSlackResult(resultPayload);

    renderLeaderboard();
    setStats();
  }

  async function storeResult(payload){
    try{
      if(firebaseReady && db){
        await addDoc(collection(db, "results"), payload);
      }else{
        saveLocalResult(payload);
        resultsCache = getLocalResults();
      }
    }catch(err){
      console.error("storeResult failed, saving local instead", err);
      saveLocalResult(payload);
      resultsCache = getLocalResults();
    }
  }

  function sendEmailResult(payload){
    $("email_name").value = payload.name;
    $("email_department").value = payload.department;
    $("email_score").value = payload.score + "%";
    $("email_status").value = payload.status;
    $("email_correct").value = String(payload.correct);
    $("email_total").value = String(payload.total);
    $("email_avg").value = String(payload.avgTime) + "s";
    $("email_time").value = formatDate(payload.time);

    try{
      $("emailForm").submit();
    }catch(err){
      console.warn("Email submit failed:", err);
    }
  }

  async function sendSlackResult(payload){
    if(!SLACK_WEBHOOK_URL) return;
    try{
      await fetch(SLACK_WEBHOOK_URL, {
        method: "POST",
        headers: { "Content-Type":"application/json" },
        body: JSON.stringify({
          text:
`📊 HOU2 PRO v4.1 Result
👤 Associate: ${payload.name}
🏷 Department: ${payload.department}
📈 Score: ${payload.score}%
✅ Correct: ${payload.correct}/${payload.total}
⏱ Avg Time: ${payload.avgTime}s
🚦 Status: ${payload.status}
🕒 Completed: ${formatDate(payload.time)}`
        })
      });
    }catch(err){
      console.warn("Slack send failed:", err);
    }
  }

  function forceCheatEnd(){
    if(!quizActive || cheatTriggered) return;
    cheatTriggered = true;
    finishQuiz("Cheating detected");
  }

  }

  async function addQuestion(){
    const cat = $("newCat").value.trim() || "General";
    const q = $("newQuestion").value.trim();
    const opt1 = $("opt1").value.trim();
    const opt2 = $("opt2").value.trim();
    const opt3 = $("opt3").value.trim();

    if(!q || !opt1 || !opt2 || !opt3){
      alert("Fill question and all 3 options.");
      return;
    }

    const payload = {
      id: "Q" + Math.random().toString(36).slice(2,9).toUpperCase(),
      cat,
      q,
      a:[opt1,opt2,opt3],
      c:0
    };

    try{
      if(firebaseReady && db){
        await setDoc(doc(db, "questions", payload.id), payload);
      }
      questionBank.unshift(payload);
      saveLocalQuestions(questionBank);
      renderQuestionList();
      setStats();
      $("newCat").value = "";
      $("newQuestion").value = "";
      $("opt1").value = "";
      $("opt2").value = "";
      $("opt3").value = "";
      alert("Question added.");
    }catch(err){
      console.error(err);
      alert("Could not add question.");
    }
  }

  async function deleteQuestion(id){
    if(!confirm("Delete this question?")) return;
    try{
      if(firebaseReady && db){
        await deleteDoc(doc(db, "questions", id));
      }
    }catch(err){
      console.warn("Firestore delete issue:", err);
    }
    questionBank = questionBank.filter(q => q.id !== id);
    saveLocalQuestions(questionBank);
    renderQuestionList();
    setStats();
  }

  function seedDefaultBank(){
    if(!confirm("Reset question bank to default questions?")) return;
    questionBank = [...defaultQuestions];
    saveLocalQuestions(questionBank);
    renderQuestionList();
    setStats();
    alert("Default bank restored.");
  }

  function renderLeaderboard(){
    const rows = [...resultsCache]
      .sort((a,b) => (b.score || 0) - (a.score || 0) || new Date(b.time||0) - new Date(a.time||0))
      .slice(0, 50);

    if(!rows.length){
      $("leaderboard").innerHTML = `<div class="muted">No results yet.</div>`;
      return;
    }

    $("leaderboard").innerHTML = rows.map((r) => {
      const meta = getResultClass(Number(r.score || 0));
      return `
        <div class="leaderboard-row">
          <div><strong>${escapeHtml(r.name || "-")}</strong><div class="small muted">${escapeHtml(r.department || "")}</div></div>
          <div>${Number(r.score || 0)}%</div>
          <div><span class="badge ${meta.badge}">${escapeHtml(r.status || meta.text)}</span></div>
          <div>${r.avgTime ?? "-"}s</div>
          <div class="small muted">${escapeHtml(formatDate(r.time))}</div>
        </div>
      `;
    }).join("");
  }

  function renderQuestionList(){
    if(!questionBank.length){
      $("questionList").innerHTML = `<div class="muted">No questions loaded.</div>`;
      return;
    }
    $("questionList").innerHTML = questionBank.map((q) => `
      <div class="question-row">
        <div>${escapeHtml(q.q)}</div>
        <div>${escapeHtml(q.cat || "General")}</div>
        <div>${escapeHtml(q.a[q.c] || "")}</div>
        <div>${q.a.length}</div>
        <div><button class="btn-red small delete-q" data-id="${escapeHtml(q.id)}">Delete</button></div>
      </div>
    `).join("");

    [...document.querySelectorAll(".delete-q")].forEach(btn => {
      btn.addEventListener("click", () => deleteQuestion(btn.dataset.id));
    });
  }
