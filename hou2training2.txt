    function byId(id){ return document.getElementById(id); }
    function normalize(v){ return String(v || '').trim().toUpperCase(); }
    function escapeHtml(value){
      return String(value)
        .replace(/&/g,'&amp;')
        .replace(/</g,'&lt;')
        .replace(/>/g,'&gt;')
        .replace(/"/g,'&quot;')
        .replace(/'/g,'&#39;');
    }
    function deepClone(v){ return JSON.parse(JSON.stringify(v)); }
    function shuffleArray(arr){
      const copy = arr.slice();
      for(let i = copy.length - 1; i > 0; i--){
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }
    function showOnly(viewId){
      ['loginView','adminView','examView','thankYouView'].forEach(id => byId(id).classList.add('hidden'));
      byId(viewId).classList.remove('hidden');
    }
    function setMsg(id,text,kind){
      const el = byId(id);
      if(!text){ el.innerHTML=''; return; }
      el.innerHTML = `<div class="msg ${kind}">${escapeHtml(text)}</div>`;
    }
    function formatDate(value){
      try{ return new Date(value).toLocaleString(); }catch{ return String(value || ''); }
    }
    function formatSeconds(value){
      const v = Math.max(0, Number(value) || 0);
      const mm = String(Math.floor(v / 60)).padStart(2,'0');
      const ss = String(v % 60).padStart(2,'0');
      return `${mm}:${ss}`;
    }

    function validateQuestion(question){
      if(!question.id || !question.cat || !question.text) throw new Error('Question requires id, category, and text.');
      if(!Array.isArray(question.options) || question.options.length !== 3) throw new Error(`${question.id} must have exactly 3 options.`);
      if(!Array.isArray(question.correctAnswers) || question.correctAnswers.length !== 1) throw new Error(`${question.id} must have exactly 1 correct answer index.`);
      const idx = question.correctAnswers[0];
      if(!Number.isInteger(idx) || idx < 0 || idx > 2) throw new Error(`${question.id} correct answer index must be 0, 1, or 2.`);
    }

    async function bootstrapLiveData(){
      await ensureDefaults();
      attachLiveSettings();
      attachLiveQuestions();
      attachLiveResults();
    }

    async function ensureDefaults(){
      const settingsSnap = await getDoc(SETTINGS_DOC);
      if(!settingsSnap.exists()){
        await setDoc(SETTINGS_DOC, DEFAULT_SETTINGS);
      }
      const qSnap = await getDocs(QUESTIONS_COLLECTION);
      if(qSnap.empty){
        for(const q of DEFAULT_QUESTIONS){
          await setDoc(doc(QUESTIONS_COLLECTION, q.id), q);
        }
      }
    }

    function attachLiveSettings(){
      onSnapshot(SETTINGS_DOC, snap => {
        settings = snap.exists() ? snap.data() : deepClone(DEFAULT_SETTINGS);
        hydrateSettingsForm();
        updateEmailFormAction();
        refreshAdminView();
      });
    }

    function attachLiveQuestions(){
      onSnapshot(query(QUESTIONS_COLLECTION, orderBy('id')), snap => {
        questions = snap.docs.map(d => d.data());
        refreshAdminView();
        if(state.role === 'associate' && state.workingQuestions.length === 0){
          // no-op until start exam
        }
      });
    }

    function attachLiveResults(){
      onSnapshot(query(RESULTS_COLLECTION, orderBy('completedAt','desc')), snap => {
        liveResults = snap.docs.map(d => ({ docId:d.id, ...d.data() }));
        refreshAdminView();
      });
    }

    async function handleLogin(){
      const loginId = normalize(byId('loginId').value);
      const password = byId('loginPassword').value;

      if(!loginId){
        setMsg('loginMsg','Enter Login ID.','bad');
        return;
      }

      if(loginId === normalize(settings.adminId)){
        if(password !== settings.adminPassword){
          setMsg('loginMsg','Invalid admin password.','bad');
          return;
        }
        resetState();
        state.loginId = loginId;
        state.role = 'admin';
        refreshAdminView();
        showOnly('adminView');
        setMsg('loginMsg','', 'ok');
        return;
      }

      const allowed = (settings.allowedLoginIds || []).map(normalize);
      if(!allowed.includes(loginId)){
        setMsg('loginMsg','Access denied. Login ID is not on the allowed list.','bad');
        return;
      }

      resetState();
      state.loginId = loginId;
      state.role = 'associate';
      startExam();
    }

    function handleLogout(){
      resetState();
      byId('loginId').value = '';
      byId('loginPassword').value = '';
      byId('feedbackBox').value = '';
      showOnly('loginView');
    }

    function refreshAdminView(){
      const results = liveResults.slice();
      const avg = results.length ? Math.round(results.reduce((a,b)=>a+(Number(b.percent)||0),0)/results.length) : 0;
      byId('statTotalResults').textContent = results.length;
      byId('statPass').textContent = results.filter(r => Number(r.percent) >= Number(settings.passingScore || 80)).length;
      byId('statFail').textContent = results.filter(r => Number(r.percent) < Number(settings.passingScore || 80)).length;
      byId('statQuestionCount').textContent = questions.length;
      byId('statAvgScore').textContent = `${avg}%`;
      renderQuestionBank();
      renderResults();
      renderLeaderboard();
    }

    async function saveSettingsFromForm(){
      const next = {
        adminId: normalize(byId('adminIdInput').value),
        adminPassword: byId('adminPasswordInput').value,
        resultEmail: byId('resultEmailInput').value.trim(),
        slackWebhook: byId('slackWebhookInput').value.trim(),
        passingScore: Math.max(0, Math.min(100, Number(byId('passMarkInput').value) || 80)),
        timerPerQuestion: Math.max(5, Math.min(300, Number(byId('timerPerQuestionInput').value) || 30)),
        allowedLoginIds: byId('associateListInput').value.split('\n').map(v => normalize(v)).filter(Boolean)
      };

      if(!next.adminId || !next.adminPassword){
        setMsg('settingsMsg','Admin ID and password are required.','bad');
        return;
      }
      if(!next.resultEmail){
        setMsg('settingsMsg','Result email is required for FormSubmit.','bad');
        return;
      }
      await setDoc(SETTINGS_DOC, next);
      setMsg('settingsMsg','Live settings saved.','ok');
    }

    function getQuestionFromForm(){
      const question = {
        id: normalize(byId('qIdInput').value),
        cat: byId('qCategoryInput').value.trim(),
        text: byId('qTextInput').value.trim(),
        options: [byId('opt1Input').value.trim(), byId('opt2Input').value.trim(), byId('opt3Input').value.trim()],

    }

    async function addQuestionFromForm(){
      try{
        const q = getQuestionFromForm();
        if(questions.some(item => normalize(item.id) === q.id)){
          setMsg('questionFormMsg','Question ID already exists. Use Update Existing Question instead.','bad');
          return;
        }
        await setDoc(doc(QUESTIONS_COLLECTION, q.id), q);
        clearQuestionForm();
        setMsg('questionFormMsg',`${q.id} added successfully.`,'ok');
      }catch(error){
        setMsg('questionFormMsg',error.message,'bad');
      }
    }

    async function updateQuestionFromForm(){
      try{
        const q = getQuestionFromForm();
        if(!questions.some(item => normalize(item.id) === q.id)){
          setMsg('questionFormMsg','Question ID not found.','bad');
          return;
        }
        await setDoc(doc(QUESTIONS_COLLECTION, q.id), q);
        clearQuestionForm();
        setMsg('questionFormMsg',`${q.id} updated successfully.`,'ok');
      }catch(error){
        setMsg('questionFormMsg',error.message,'bad');
      }
    }

    function renderLeaderboard(){
      const bestByLogin = {};
      for(const r of liveResults){
        const key = normalize(r.loginId);
        if(!bestByLogin[key] || Number(r.percent) > Number(bestByLogin[key].percent)){
          bestByLogin[key] = r;
        }
      }
      const rows = Object.values(bestByLogin)
        .sort((a,b) => Number(b.percent) - Number(a.percent) || new Date(a.completedAt) - new Date(b.completedAt))
        .slice(0,10);

      byId('leaderboardList').innerHTML = rows.length ? rows.map((r,idx) => `
        <div class="leader-row">
          <div>
            <strong>#${idx+1} ${escapeHtml(r.loginId)}</strong>
            <div class="small">Weak Area: ${escapeHtml(r.weakArea || 'None')}</div>
          </div>
          <div><span class="chip">${escapeHtml(r.percent)}%</span></div>
        </div>
      `).join('') : '<div class="small">No leaderboard data yet.</div>';
    }

    function buildWorkingQuestions(){
      return shuffleArray(questions).map(q => {
        const indexedOptions = q.options.map((text, index) => ({ text, originalIndex:index }));
        const shuffledOptions = shuffleArray(indexedOptions);
        return {
          id:q.id,
          cat:q.cat,
          text:q.text,
          options:shuffledOptions.map(o => o.text),
          correctAnswers:[shuffledOptions.findIndex(o => q.correctAnswers.includes(o.originalIndex))]
        };
      });
    }

    function startExam(){
      state.workingQuestions = buildWorkingQuestions();
      state.startAt = new Date().toISOString();
      state.qIndex = 0;
      state.score = 0;
      state.answers = [];
      state.catMiss = {};
      state.antiCheatFlags = 0;
      state.tabSwitchCount = 0;
      setMsg('antiCheatMsg','', 'warn');
      showOnly('examView');
      renderQuestion();
    }

    function renderQuestion(){
      const q = state.workingQuestions[state.qIndex];
      byId('examHeader').textContent = `Associate Test - ${state.loginId}`;
      byId('examMeta').textContent = `Category: ${q.cat} | Randomized order active | Anti-cheat enabled`;
      byId('questionCounter').textContent = `Question ${state.qIndex + 1} / ${state.workingQuestions.length}`;
      byId('progressBar').style.width = `${((state.qIndex + 1) / state.workingQuestions.length) * 100}%`;
      byId('questionText').textContent = `${q.id} - ${q.text}`;
      byId('questionOptions').innerHTML = q.options.map((opt, i) => `
        <label class="option">
          <input type="radio" name="currentQuestionOption" value="${i}" onchange="window.submitCurrentAnswer(${i})">
          <div><strong>${String.fromCharCode(65+i)}.</strong> ${escapeHtml(opt)}</div>
        </label>
      `).join('');
      startQuestionTimer();
    }

    function startQuestionTimer(){
      clearInterval(examTimer);
      examSecondsLeft = Number(settings.timerPerQuestion || 30);
      byId('timerChip').textContent = `Time: ${formatSeconds(examSecondsLeft)}`;
      examTimer = setInterval(() => {
        examSecondsLeft -= 1;
        byId('timerChip').textContent = `Time: ${formatSeconds(examSecondsLeft)}`;
        if(examSecondsLeft <= 0){
          clearInterval(examTimer);
          window.submitCurrentAnswer(null, true);
        }
      }, 1000);
    }

    function submitCurrentAnswer(selectedIndex, timedOut = false){
      if(isSubmittingAnswer) return;
      isSubmittingAnswer = true;
      clearInterval(examTimer);

      const q = state.workingQuestions[state.qIndex];
      const safeIndex = Number.isInteger(selectedIndex) ? selectedIndex : -1;
      const isCorrect = q.correctAnswers.includes(safeIndex);

      if(isCorrect){
        state.score += 1;
      }else{
        state.catMiss[q.cat] = (state.catMiss[q.cat] || 0) + 1;
      }

      state.answers.push({
        id:q.id,
        category:q.cat,
        question:q.text,
        selected: safeIndex >= 0 ? q.options[safeIndex] : (timedOut ? 'No answer / timed out' : 'No answer'),
        correct: q.correctAnswers.map(i => q.options[i]).join(' | '),
        isCorrect,
        timedOut
      });

      state.qIndex += 1;
      if(state.qIndex >= state.workingQuestions.length){
        finishExam().finally(() => { isSubmittingAnswer = false; });
        return;
      }
      renderQuestion();
      isSubmittingAnswer = false;
    }

    async function finishExam(){
      const percent = Math.round((state.score / state.workingQuestions.length) * 100);
      const weak = Object.entries(state.catMiss).sort((a,b) => b[1]-a[1])[0];
      const weakArea = weak ? weak[0] : 'None';
      const categoryBreakdown = deepClone(state.catMiss);
      const completedAt = new Date().toISOString();

      const result = {
        loginId: state.loginId,
        percent,
        weakArea,
        categoryBreakdown,
        breakdown: state.answers.filter(a => !a.isCorrect).map(a => ({
          id:a.id, category:a.category, question:a.question, selected:a.selected, correct:a.correct
        })),
        completedAt,
        feedback:'',
        antiCheatFlags: state.antiCheatFlags,
        tabSwitchCount: state.tabSwitchCount,
        startedAt: state.startAt,
        totalQuestions: state.workingQuestions.length
      };

      const ref = await addDoc(RESULTS_COLLECTION, { ...result, createdAt: serverTimestamp() });
      currentSessionId = ref.id;

      sendSlackAlert(result);
      sendEmailResult(result);
      byId('finalSummary').textContent = `Login ID: ${result.loginId}\nScore: ${result.percent}%\nResult: ${result.percent >= Number(settings.passingScore || 80) ? 'PASS' : 'FAIL'}\nWeak Area: ${result.weakArea}\nAnti-Cheat Flags: ${result.antiCheatFlags}\nTab Switches: ${result.tabSwitchCount}`;
      byId('feedbackBox').value = '';
      showOnly('thankYouView');
    }

    async function submitFeedbackAndReset(){
      const feedback = byId('feedbackBox').value.trim();
      if(currentSessionId){
        await updateDoc(doc(RESULTS_COLLECTION, currentSessionId), { feedback });
      }
      handleLogout();
    }

    function sendSlackAlert(result){
      if(!settings.slackWebhook) return;
      const breakdownText = result.breakdown.length
        ? result.breakdown.map(b => [
            `${b.id} (${b.category})`,
            `Question: ${b.question}`,
            `Selected: ${b.selected}`,
            `Correct: ${b.correct}`
          ].join('\n')).join('\n----------------------\n')
        : 'No mistakes';

      const categoryText = Object.keys(result.categoryBreakdown).length
        ? Object.entries(result.categoryBreakdown).map(([k,v]) => `${k}: ${v}`).join(' | ')
        : 'No misses';

      const message = [
        'HOU2 LIVE TEST RESULT',
        '',
        `Login ID: ${result.loginId}`,
        `Score: ${result.percent}%`,
        `Result: ${result.percent >= Number(settings.passingScore || 80) ? 'PASS' : 'FAIL'}`,
        `Weak Area: ${result.weakArea}`,
        `Category Breakdown: ${categoryText}`,
        `Anti-Cheat Flags: ${result.antiCheatFlags}`,
        `Tab Switches: ${result.tabSwitchCount}`,
        '',
        '--- Mistakes ---',
        breakdownText
      ].join('\n');

      fetch(settings.slackWebhook, {
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body:JSON.stringify({ text: message })
      }).catch(err => console.error('Slack send failed:', err));
    }

    function sendEmailResult(result){
      try{
        if(!settings.resultEmail || settings.resultEmail === 'YOUR_EMAIL_HERE') return;
        const breakdownText = result.breakdown.length
          ? result.breakdown.map(b => `${b.id} (${b.category})\nQ: ${b.question}\nSelected: ${b.selected}\nCorrect: ${b.correct}`).join('\n----------------------\n')
          : 'No mistakes';

        byId('email_loginId').value = result.loginId;
        byId('email_score').value = `${result.percent}%`;
        byId('email_result').value = result.percent >= Number(settings.passingScore || 80) ? 'PASS' : 'FAIL';
        byId('email_weakArea').value = result.weakArea;
        byId('email_breakdown').value = breakdownText;
        byId('email_subject').value = `HOU2 V5 Result - ${result.loginId} - ${result.percent >= Number(settings.passingScore || 80) ? 'PASS' : 'FAIL'} - ${result.percent}%`;
        updateEmailFormAction();
        byId('emailForm').submit();
      }catch(err){
        console.error('Email send failed:', err);
      }
    }

    function startSystem(){
      if(introHasStarted) return;
      introHasStarted = true;
      const intro = byId('introScreen');
      const loading = byId('loadingScreen');
      const appRoot = byId('appRoot');
      intro.style.animation = 'fadeOut 0.8s forwards';
      setTimeout(() => {
        intro.remove();
        loading.classList.remove('hidden');
        setTimeout(() => {
          loading.style.animation = 'fadeOut 0.8s forwards';
          setTimeout(() => {
            loading.remove();
            appRoot.classList.remove('hidden');
          }, 800);
        }, 1200);
      }, 700);
    }

      const intro = byId('introScreen');
      const loading = byId('loadingScreen');
      if(intro) intro.remove();
      if(loading) loading.remove();
      console.error(error);
    });
